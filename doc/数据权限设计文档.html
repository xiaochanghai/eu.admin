1. 设计概述

1.1 核心思想

参考 SmRoleModule 的树形勾选模式，实现数据权限的树形配置界面。

1.2 设计对比

特性 SmRoleModule（菜单权限） 数据权限（本次设计）
树形结构 模块树（父子关系） 集团-公司树（两层结构）
键值编码 ModuleId, CommonOption_*, functionPrivileges_* All, Group_{id}, Company_{id}
前端渲染 Ant Design Tree + Checkbox Ant Design Tree + Checkbox
保存方式 提交 checkedKeys 列表 提交 checkedKeys 列表
后端处理 解析键值，保存到 SmRoleModule/SmRoleFunction 解析键值，展开为具体公司 ID，保存到 SmRoleDataScope

1.3 树形结构示例

All (全部)
├── Group_A (集团 A)
│   ├── Company_A1 (公司 A1)
│   ├── Company_A2 (公司 A2)
│   └── Company_A3 (公司 A3)
├── Group_B (集团 B)
│   ├── Company_B1 (公司 B1)
│   └── Company_B2 (公司 B2)
└── Group_C (集团 C)
    └── Company_C1 (公司 C1)

2. 核心设计模式

2.1 键值编码规则

键值格式 含义 后端展开逻辑
All 全部数据 查询所有集团下的所有公司，插入所有公司 ID
Group_{groupId} 指定集团 查询该集团下的所有公司，插入所有公司 ID
Company_{companyId} 指定公司 直接插入该公司 ID

2.2 数据流程

4. 树形结构构建

4.1 后端 API：获取树形结构

/// <summary>
/// 获取所有数据权限树（集团-公司树形结构）
/// 直接返回集团列表，每个集团包含下属公司
/// </summary>
[HttpGet("GetAllDataScopeTree")]
public async Task<ServiceResult<List<DataScopeTree>>> GetAllDataScopeTree()
{
    // 1. 查询所有集团
    var groups = await Db.Queryable<SmGroup>()
        .Where(x => x.IsDeleted == false)
        .OrderBy(x => x.GroupCode)
        .ToListAsync();

    // 2. 查询所有公司
    var companies = await Db.Queryable<SmCompany>()
        .Where(x => x.IsDeleted == false)
        .OrderBy(x => x.CompanyCode)
        .ToListAsync();

    // 3. 构建树形结构（直接返回集团列表）
    var tree = new List<DataScopeTree>();
    foreach (var group in groups)
    {
        var groupNode = new DataScopeTree
        {
            key = group.ID.ToString(),  // 直接用 GUID
            title = $"[集团] {group.GroupName}",
            value = group.ID.ToString(),
            isLeaf = false,
            children = new List<DataScopeTree>()
        };

        var groupCompanies = companies.Where(c => c.GroupId == group.ID).ToList();
        foreach (var company in groupCompanies)
        {
            groupNode.children.Add(new DataScopeTree
            {
                key = company.ID.ToString(),  // 直接用 GUID
                title = company.CompanyName,
                value = company.ID.ToString(),
                isLeaf = true
            });
        }

        tree.Add(groupNode);
    }

    return Success(tree);
}

// DataScopeSet.tsx
import { Tabs, Card, Button, Checkbox, Collapse } from 'antd';
import http from '@/api';

const DATA_SCOPE_API_URL = "/api/SmRoleDataScope";

/**
 * 数据权限项类型定义
 */
interface DataScopeItem {
  key: string;      // 集团或公司的 GUID
  title: string;
  isLeaf?: boolean; // 公司节点为 true
  children?: DataScopeItem[];
}

const DataScopeSet: React.FC<{ roleId: string }> = ({ roleId }) => {
  const [dataScopes, setDataScopes] = useState<DataScopeItem[]>([]);
  const [checkedDataScopeKeys, setCheckedDataScopeKeys] = useState<string[]>([]);

  // 加载树形结构（后端直接返回集团列表）
  const fetchAllDataScopeTree = async () => {
    const { Data, Success } = await http.get<any>(`${DATA_SCOPE_API_URL}/GetAllDataScopeTree`);
    if (Success) setDataScopes(Data || []);  // 直接使用 Data
  };

  // 加载角色已有权限（直接返回公司 ID 列表）
  const fetchRoleDataScope = async () => {
    const { Data, Success } = await http.get<any>(`${DATA_SCOPE_API_URL}/GetRoleDataScope/${roleId}`);
    if (Success) setCheckedDataScopeKeys(Data || []);
  };

  // 保存权限（直接发送公司 ID 列表）
  const handleSave = async () => {
    await http.post(`${DATA_SCOPE_API_URL}/UpdateDataScope/${roleId}`, checkedDataScopeKeys);
  };

  return (
    <div>
      <Collapse items={...} />
      <Button onClick={handleSave}>保存</Button>
    </div>
  );
};

5. 权限保存逻辑

5.1 后端 API：更新数据权限

/// <summary>
/// 更新数据权限请求模型
/// </summary>
public class UpdateDataScopeRequest
{
    /// <summary>
    /// 权限键值列表
    /// </summary>
    public List<string> ScopeKeys { get; set; }

    /// <summary>
    /// 期望版本号（用于并发控制）
    /// </summary>
    public int? ExpectedVersion { get; set; }
}

/// <summary>
/// 更新角色数据权限
/// </summary>
[HttpPost("UpdateDataScope/{roleId}")]
public async Task<ServiceResult> UpdateDataScope(Guid roleId, [FromBody] UpdateDataScopeRequest request)
{
    try
    {
        // 0. 输入验证
        if (request == null || request.ScopeKeys == null)
        {
            return Failed("请求参数不能为空");
        }

        // 验证键值格式
        foreach (var key in request.ScopeKeys)
        {
            if (string.IsNullOrWhiteSpace(key))
            {
                return Failed("键值不能为空");
            }

            if (key != "All" && !key.StartsWith("Group_") && !key.StartsWith("Company_"))
            {
                return Failed($"无效的键值格式: {key}");
            }

            // 验证 GUID 格式
            if (key.StartsWith("Group_") || key.StartsWith("Company_"))
            {
                var idPart = key.Substring(key.IndexOf('_') + 1);
                if (!Guid.TryParse(idPart, out _))
                {
                    return Failed($"无效的 GUID 格式: {key}");
                }
            }
        }

        // 1. 并发控制检查
        var role = await Db.Queryable<SmRoles>()
            .Where(x => x.ID == roleId)
            .FirstAsync();

        if (role == null)
            return Failed("角色不存在");

        // 检查版本号是否匹配
        if (request.ExpectedVersion.HasValue && role.DataScopeVersion != request.ExpectedVersion.Value)
        {
            return Failed($"数据已被其他用户修改（当前版本: {role.DataScopeVersion}，您提交的版本: {request.ExpectedVersion.Value}），请刷新页面后重试");
        }

        // 2. 获取旧值（用于审计日志）
        var oldValue = await GetRoleDataScope(roleId);

        // 3. 开启事务
        await Db.Ado.BeginTranAsync();

        try
        {
            // 4. 删除旧权限
            await Db.Deleteable<SmRoleDataScope>()
                .Where(x => x.SmRoleId == roleId)
                .ExecuteCommandAsync();

            // 5. 插入新权限（直接插入公司 ID）
            if (companyIds.Any())
            {
                var newScopes = companyIds.Select(companyId => new SmRoleDataScope
                {
                    SmRoleId = roleId,
                    CompanyId = companyId,
                    CreatedBy = UserContext.Current.User_Id,
                    CreatedTime = DateTime.Now
                }).ToList();

                await Db.Insertable(newScopes).ExecuteCommandAsync();
            }

            // 7. 更新角色版本号
            role.DataScopeVersion++;
            await Db.Updateable(role)
                .UpdateColumns(x => x.DataScopeVersion)
                .ExecuteCommandAsync();

            // 8. 提交事务
            await Db.Ado.CommitTranAsync();

            // 9. 记录审计日志
            await AddAuditLog(new SmRoleDataScopeAudit
            {
                SmRoleId = roleId,
                Action = "Update",
                OldValue = JsonConvert.SerializeObject(oldValue.Data),
                NewValue = JsonConvert.SerializeObject(request.ScopeKeys),
                OperatedBy = UserContext.Current.User_Id,
                OperatedTime = DateTime.Now,
                IpAddress = HttpContextExtension.GetUserIp(HttpContext.Current),
                UserAgent = HttpContext.Current?.Request?.Headers["User-Agent"],
                IsSuccess = true,
                Reason = "更新数据权限"
            });

            // 10. 清除受影响用户的缓存
            await ClearAffectedUsersCache(roleId);

            return Success("更新成功");
        }
        catch (Exception ex)
        {
            await Db.Ado.RollbackTranAsync();
            throw;
        }
    }
    catch (Exception ex)
    {
        return Failed($"更新失败: {ex.Message}");
    }
}

/// <summary>
/// 展开键值为具体的公司 ID 列表（优化版：批量查询）
/// </summary>
private async Task<List<Guid>> ExpandScopeKeysOptimized(List<string> scopeKeys)
{
    var companyIds = new HashSet<Guid>();

    // 分类键值
    var groupIds = new List<Guid>();
    var companyIdsDirect = new List<Guid>();
    bool needAll = false;

    foreach (var key in scopeKeys)
    {
        if (key == "All")
        {
            needAll = true;
            break;
        }
        else if (key.StartsWith("Group_"))
        {
            var groupId = Guid.Parse(key.Replace("Group_", ""));
            groupIds.Add(groupId);
        }
        else if (key.StartsWith("Company_"))
        {
            var companyId = Guid.Parse(key.Replace("Company_", ""));
            companyIdsDirect.Add(companyId);
        }
    }

    // 批量查询
    if (needAll)
    {
        // 查询所有公司
        var allCompanies = await Db.Queryable<SmCompany>()
            .Where(x => x.IsDeleted == false)
            .Select(x => x.ID)
            .ToListAsync();

        companyIds.UnionWith(allCompanies);
    }
    else
    {
        // 查询指定集团下的所有公司
        if (groupIds.Any())
        {
            var groupCompanies = await Db.Queryable<SmCompany>()
                .Where(x => groupIds.Contains(x.GroupId) && x.IsDeleted == false)
                .Select(x => x.ID)
                .ToListAsync();

            companyIds.UnionWith(groupCompanies);
        }

        // 添加直接指定的公司
        companyIds.UnionWith(companyIdsDirect);
    }

    return companyIds.ToList();
}

5.2 后端 API：获取角色数据权限

/// <summary>
/// 获取角色的数据权限（返回键值列表）
/// </summary>
[HttpGet("GetRoleDataScope/{roleId}")]
public async Task<ServiceResult<List<string>>> GetRoleDataScope(Guid roleId)
{
    // 查询该角色的所有公司权限
    var scopes = await Db.Queryable<SmRoleDataScope>()
        .Where(x => x.SmRoleId == roleId)
        .Select(x => x.CompanyId)
        .ToListAsync();

    // 如果没有权限，返回空列表
    if (!scopes.Any())
    {
        return Success(new List<string>());
    }

    // 查询所有集团
    var groups = await Db.Queryable<SmGroup>()
        .Where(x => x.IsDeleted == false)
        .ToListAsync();

    // 查询所有未删除的公司
    var companies = await Db.Queryable<SmCompany>()
        .Where(x => x.IsDeleted == false)
        .ToListAsync();

    var keys = new List<string>();

    // 判断是否是"全部"（所有未删除的公司都有权限）
    if (scopes.Count == companies.Count)
    {
        keys.Add("All");
    }
    else
    {
        // 判断是否可以合并为集团
        foreach (var group in groups)
        {
            // 获取该集团下所有未删除的公司ID
            var groupCompanies = companies
                .Where(c => c.GroupId == group.ID)
                .Select(c => c.ID)
                .ToList();

            // 如果集团下没有公司，跳过
            if (!groupCompanies.Any())
            {
                continue;
            }

            // 获取该角色在该集团下有权限的公司
            var scopedGroupCompanies = scopes
                .Where(s => groupCompanies.Contains(s))
                .ToList();

            // 如果该集团下所有公司都有权限，则添加集团键值
            if (scopedGroupCompanies.Count == groupCompanies.Count)
            {
                keys.Add($"Group_{group.ID}");
            }
            else if (scopedGroupCompanies.Any())
            {
                // 否则添加具体的公司键值
                foreach (var companyId in scopedGroupCompanies)
                {
                    keys.Add($"Company_{companyId}");
                }
            }
        }
    }

    return Success(keys);
}

6. 权限查询逻辑

6.0 无权限场景说明（重要）

⚠️ 默认行为定义： 当用户没有任何数据权限时，无法访问任何数据（拒绝访问）。
场景 CompanyIds 行为 说明
用户没有角色 [] (空列表) 拒绝访问 用户未分配任何角色
用户有角色但无数据权限 [] (空列表) 拒绝访问 角色未配置数据权限
用户有数据权限 [guid1, guid2, ...] 允许访问指定公司 只能访问列表中的公司数据

6.0.1 代码实现

/// <summary>
/// 获取用户的实际数据范围
/// </summary>
public async Task<ServiceResult<UserDataScopeModel>> GetUserDataScope(Guid userId)
{
    var model = new UserDataScopeModel();

    // 1. 获取用户的所有角色
    var userRoles = await Db.Queryable<SmUserRole>()
        .Where(x => x.SmUserId == userId)
        .Select(x => x.SmRoleId)
        .ToListAsync();

    // 场景1：用户没有角色 → 返回空列表（拒绝访问）
    if (!userRoles.Any())
    {
        return Success(model); // model.CompanyIds = []
    }

    // 2. 获取这些角色的数据权限（公司 ID 列表）
    var companyIds = await Db.Queryable<SmRoleDataScope>()
        .Where(x => userRoles.Contains(x.SmRoleId))
        .Select(x => x.CompanyId)
        .Distinct()
        .ToListAsync();

    // 场景2：有角色但无数据权限 → 返回空列表（拒绝访问）
    // 场景3：有数据权限 → 返回公司ID列表
    model.CompanyIds = companyIds;

    return Success(model);
}

/// <summary>
/// 构建查询过滤表达式
/// </summary>
public static Expression<Func<T, bool>> BuildFilterExpression<T>(UserDataScopeModel scope)
    where T : class, IDataScopeEntity
{
    // 如果没有权限，返回 false（拒绝访问）
    if (!scope.CompanyIds.Any())
    {
        return x => false; // 永远返回 false，查询结果为空
    }

    // 有权限，返回 IN 条件
    return x => scope.CompanyIds.Contains(x.CompanyId);
}

/// <summary>
/// 检查用户是否有权限访问指定数据
/// </summary>
public static async Task<bool> HasPermission<T>(Guid userId, T entity)
    where T : class, IDataScopeEntity
{
    var scope = await GetUserDataScope(userId);

    // 如果没有权限，直接返回 false
    if (!scope.CompanyIds.Any())
    {
        return false;
    }

    // 检查是否在权限列表中
    return scope.CompanyIds.Contains(entity.CompanyId);
}

// 示例2：检查单个订单的访问权限
public async Task<SdOrder> GetOrderById(Guid orderId)
{
    var order = await Db.Queryable<SdOrder>()
        .Where(x => x.ID == orderId)
        .FirstAsync();

    if (order == null)
    {
        throw new Exception("订单不存在");
    }

    // 检查数据权限
    var hasPermission = await DataScopeHelper.HasPermission(
        UserContext.Current.User_Id.Value,
        order
    );

    if (!hasPermission)
    {
        throw new Exception("您没有权限访问此订单");
    }

    return order;
}
💡 设计原则：
  • 最小权限原则：默认拒绝所有访问，除非明确授权
  • 安全性优先：无权限时返回空结果，而不是抛出异常（避免信息泄露）
  • 一致性：所有数据权限检查都遵循相同的规则
  • 可审计：可以通过日志追踪权限检查的结果

/// <summary>
/// 数据权限验证工具类
/// </summary>
public static class DataScopeHelper
{
    private static readonly ISmRoleDataScopeServices _service =
        App.GetService<ISmRoleDataScopeServices>();

    /// <summary>
    /// 获取用户的实际数据范围（带缓存）
    /// </summary>
    public static async Task<UserDataScopeModel> GetUserDataScope(Guid userId)
    {
        var cacheKey = $"UserDataScope_{userId}";
        var cached = await Redis.GetAsync<UserDataScopeModel>(cacheKey);

        if (cached != null)
            return cached;

        var result = await _service.GetUserDataScope(userId);
        if (result.Success)
        {
            await Redis.SetAsync(cacheKey, result.Data, TimeSpan.FromMinutes(30));
            return result.Data;
        }

        return new UserDataScopeModel();
    }

    /// <summary>
    /// 构建查询过滤表达式
    /// </summary>
    public static Expression<Func<T, bool>> BuildFilterExpression<T>(UserDataScopeModel scope)
        where T : class, IDataScopeEntity
    {
        if (scope.CompanyIds.Any())
        {
            return x => scope.CompanyIds.Contains(x.CompanyId);
        }

        return x => false;
    }

    /// <summary>
    /// 应用数据权限过滤到查询
    /// </summary>
    public static async Task<ISugarQueryable<T>> ApplyDataScope<T>(
        ISugarQueryable<T> query, Guid userId)
        where T : class, IDataScopeEntity
    {
        var scope = await GetUserDataScope(userId);
        var filter = BuildFilterExpression<T>(scope);
        return query.Where(filter);
    }

    /// <summary>
    /// 检查用户是否有权限访问指定数据
    /// </summary>
    public static async Task<bool> HasPermission<T>(Guid userId, T entity)
        where T : class, IDataScopeEntity
    {
        var scope = await GetUserDataScope(userId);
        return scope.CompanyIds.Contains(entity.CompanyId);
    }

    /// <summary>
    /// 清除用户数据范围缓存
    /// </summary>
    public static async Task ClearCache(Guid userId)
    {
        var cacheKey = $"UserDataScope_{userId}";
        await Redis.RemoveAsync(cacheKey);
    }
}

8.0 缓存失效机制（重要）

⚠️ 缓存一致性： 数据权限缓存在以下场景必须清除，否则会导致权限不生效：
场景 需要清除的缓存 实现位置
更新角色数据权限 该角色所有用户的缓存 SmRoleDataScopeServices.UpdateDataScope
删除公司 使用该公司权限的所有用户缓存 SmCompanyServices.Delete
用户角色变更 该用户的缓存 SmUserRoleServices（需新增）
删除角色 该角色所有用户的缓存 SmRolesServices（需新增）
删除用户 该用户的缓存 SmUsersServices（需新增）

8.0.1 SmUserRole 变更时清除缓存

public class SmUserRoleServices : BaseServices<SmUserRole, ...>
{
    /// <summary>
    /// 分配角色给用户后，清除用户缓存
    /// </summary>
    public async Task AssignRoleToUser(Guid userId, Guid roleId)
    {
        // 1. 分配角色
        var userRole = new SmUserRole
        {
            SmUserId = userId,
            SmRoleId = roleId
        };
        await Add(userRole);

        // 2. 清除用户的数据权限缓存
        await DataScopeHelper.ClearCache(userId);
    }

    /// <summary>
    /// 移除用户角色后，清除用户缓存
    /// </summary>
    public async Task RemoveRoleFromUser(Guid userId, Guid roleId)
    {
        // 1. 移除角色
        await Delete(x => x.SmUserId == userId && x.SmRoleId == roleId);

        // 2. 清除用户的数据权限缓存
        await DataScopeHelper.ClearCache(userId);
    }

    /// <summary>
    /// 批量更新用户角色后，清除用户缓存
    /// </summary>
    public async Task UpdateUserRoles(Guid userId, List<Guid> roleIds)
    {
        // 1. 删除旧角色
        await Delete(x => x.SmUserId == userId);

        // 2. 添加新角色
        var newUserRoles = roleIds.Select(roleId => new SmUserRole
        {
            SmUserId = userId,
            SmRoleId = roleId
        }).ToList();

        if (newUserRoles.Any())
            await Add(newUserRoles);

        // 3. 清除用户的数据权限缓存
        await DataScopeHelper.ClearCache(userId);
    }
}

public class SmRolesServices : BaseServices<SmRoles, ...>
{
    public override async Task<bool> Delete(Guid[] ids)
    {
        // 1. 获取受影响的用户
        var affectedUserIds = await Db.Queryable<SmUserRole>()
            .Where(x => ids.Contains(x.SmRoleId))
            .Select(x => x.SmUserId)
            .Distinct()
            .ToListAsync();

        // 2. 删除角色
        var result = await base.Delete(ids);

            await Db.Deleteable<SmRoleDataScope>()
                .Where(x => ids.Contains(x.SmRoleId))
                .ExecuteCommandAsync();

            foreach (var userId in affectedUserIds)
            {
                await DataScopeHelper.ClearCache(userId);
            }

            Utility.ReInitCache(Db);
        }

        return result;
    }
}

8.1 SmCompany 删除时级联清理

public class SmCompanyServices : BaseServices<SmCompany, SmCompanyDto, InsertSmCompanyInput, EditSmCompanyInput>
{
    public override async Task<bool> Delete(Guid[] ids)
    {
        var result = await base.Delete(ids);

        if (result)
        {
            // 级联删除权限记录
            await Db.Deleteable<SmRoleDataScope>()
                .Where(x => ids.Contains(x.CompanyId))
                .ExecuteCommandAsync();

            // 清除相关用户缓存
            await ClearAffectedUsersCache(ids);

            Utility.ReInitCache(Db);
        }

        return result;
    }

    private async Task ClearAffectedUsersCache(Guid[] companyIds)
    {
        var roleIds = await Db.Queryable<SmRoleDataScope>()
            .Where(x => companyIds.Contains(x.CompanyId))
            .Select(x => x.SmRoleId)
            .Distinct()
            .ToListAsync();

        if (!roleIds.Any()) return;

        var userIds = await Db.Queryable<SmUserRole>()
            .Where(x => roleIds.Contains(x.SmRoleId))
            .Select(x => x.SmUserId)
            .Distinct()
            .ToListAsync();

        foreach (var userId in userIds)
        {
            await DataScopeHelper.ClearCache(userId);
        }
    }
}

8.2 SmGroup 删除时的处理

💡 设计说明：
  • 删除集团时，不需要删除 SmRoleDataScope 记录，因为权限是基于公司ID的
  • 删除集团后，GetAllDataScopeTree 会自动过滤已删除的集团，树形结构不会显示已删除的集团
  • 如果集团下有公司，建议先删除或转移公司，再删除集团
public class SmGroupServices : BaseServices<SmGroup, SmGroupDto, InsertSmGroupInput, EditSmGroupInput>
{
    public override async Task<bool> Delete(Guid[] ids)
    {
        // 检查集团下是否还有公司
        var hasCompanies = await Db.Queryable<SmCompany>()
            .Where(x => ids.Contains(x.GroupId) && x.IsDeleted == false)
            .AnyAsync();

        if (hasCompanies)
        {
            return Failed("该集团下还有公司，请先删除或转移公司后再删除集团");
        }

        // 删除集团
        var result = await base.Delete(ids);

        if (result)
        {
            Utility.ReInitCache(Db);
        }

        return result;
    }
}
⚠️ 注意： 删除集团后，如果之前有角色配置了该集团下公司的权限，这些权限记录仍然保留。但由于公司已被删除或转移，这些权限在查询时会自动被过滤掉（因为 GetUserDataScope 只返回有效的公司ID）。

// handleCheckChange - 勾选集团节点
const handleCheckChange = (e: any, item: DataScopeItem): void => {
    const newCheckedKeys = [...checkedKeys];

    // 移除当前项及其所有子项
    const removeCheckedKeys = (keys: string[], parent: DataScopeItem): void => {
        const index = keys.findIndex(x => x === parent.key);
        if (index !== -1) keys.splice(index, 1);

        if (parent.children)
            parent.children.forEach((child: DataScopeItem) => {
                removeCheckedKeys(keys, child);
            });
    };

    // 添加当前项及其所有子项
    const addCheckedKeys = (keys: string[], parent: DataScopeItem): void => {
        keys.push(parent.key);

        if (parent.children)
            parent.children.forEach((child: DataScopeItem) => {
                addCheckedKeys(keys, child);
            });
    };

    // 先移除所有相关项
    removeCheckedKeys(newCheckedKeys, item);

    // 如果选中，则添加所有相关项
    if (e.target.checked) addCheckedKeys(newCheckedKeys, item);

    setCheckedKeys(newCheckedKeys);
};
2. 勾选公司时的处理
// handleGroupChange - 勾选公司节点（CheckboxGroup）
const handleGroupChange = (checkedList: string[], parent: DataScopeItem): void => {
    const newCheckedKeys = [...checkedKeys];

    // 移除所有子项
    parent.children?.forEach((item: DataScopeItem) => {
        const index = newCheckedKeys.findIndex(x => x === item.key);
        if (index !== -1) newCheckedKeys.splice(index, 1);
    });

    // 如果所有子项都选中，则添加父项（集团）
    if (checkedList.length === parent.children?.length) {
        newCheckedKeys.push(parent.key);
    } else {
        // 否则移除父项
        const parentIndex = newCheckedKeys.findIndex(x => x === parent.key);
        if (parentIndex !== -1) newCheckedKeys.splice(parentIndex, 1);
    }

    // 添加选中的子项
    setCheckedKeys([...newCheckedKeys, ...checkedList]);
};
3. 半选状态计算
// calculateIndeterminate - 计算半选状态
const calculateIndeterminate = (item: DataScopeItem): boolean => {
    if (checkedKeys.length === 0 || !item.children) return false;

    let checkedCount = 0;
    let totalCount = 0;

    const countCheckedItems = (parent: DataScopeItem): void => {
        if (parent.children) {
            totalCount += parent.children.length;
            parent.children.forEach((child: DataScopeItem) => {
                if (some(checkedKeys, child.key)) checkedCount++;
                if (child.children) countCheckedItems(child);
            });
        }
    };

    countCheckedItems(item);

    // 部分选中时为半选状态
    return checkedCount !== 0 && checkedCount !== totalCount;
};

// handleSave - 保存权限配置（带完整错误处理）
const handleSave = async (): Promise<void> => {
    message.loading("数据提交中...", 0);
    setLoading(true);
    NProgress.start();

    try {
        // 提交 checkedKeys 列表到后端
        // 例如: ["Group_A", "Company_B1", "Company_B2"]
        const { Message, Success } = await http.post<any>(
            `${API_URL}/UpdateDataScope/${id}`,
            checkedKeys
        );

        message.destroy();

        if (Success) {
            message.success(Message || "保存成功");
            // 可选：重新加载数据以刷新界面
            // await fetchRoleDataScope();
        } else {
            message.error(Message || "保存失败，请重试");
        }
    } catch (error: any) {
        message.destroy();

        // 处理不同类型的错误
        if (error.response) {
            // 服务器返回错误状态码
            const { status, data } = error.response;

            if (status === 409) {
                // 并发冲突
                message.error(data?.Message || "数据已被其他用户修改，请刷新页面后重试");
            } else if (status === 400) {
                // 请求参数错误
                message.error(data?.Message || "请求参数错误，请检查输入");
            } else if (status === 401 || status === 403) {
                // 权限不足
                message.error("您没有权限执行此操作");
            } else if (status === 500) {
                // 服务器内部错误
                message.error("服务器内部错误，请联系管理员");
            } else {
                // 其他错误
                message.error(data?.Message || `请求失败 (${status})，请重试`);
            }
        } else if (error.request) {
            // 请求已发送但没有收到响应
            message.error("网络请求失败，请检查网络连接");
        } else {
            // 其他错误
            message.error(error.message || "保存失败，请重试");
        }

        // 记录错误日志（可选）
        console.error("保存数据权限失败:", error);
    } finally {
        setLoading(false);
        NProgress.done();
    }
};
💡 错误处理最佳实践：

  • 并发冲突（409）时提示用户刷新页面
  • 网络错误时提示检查网络连接
  • 记录错误日志便于调试
  • 在 finally 中重置加载状态
